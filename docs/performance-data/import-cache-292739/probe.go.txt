package pysolate

import("context";"encoding/json";"fmt";"os";"testing";"time")

func TestImportStudy(t *testing.T){
 wasm,err:=readGuestArtifact();if err!=nil{t.Fatal(err)}
 ctx:=context.Background();r,err:=NewPreparedCOW(ctx,wasm,nil,COWOptions{DataImage:true});if err!=nil{t.Fatal(err)};defer r.Close(ctx)
 cases:=[]struct{name,body string;want int64}{
 {"csv-json",`import csv, io, json
import_done = time.perf_counter_ns()
rows = list(csv.DictReader(io.StringIO('name,value\na,1\nb,2\nc,3\n')))
value = sum(int(row['value']) for row in json.loads(json.dumps(rows)))`,6},
 {"numpy-small",`import numpy as np
import_done = time.perf_counter_ns()
value = int(np.arange(64).sum())`,2016},
 {"numpy-matrix",`import numpy as np
import_done = time.perf_counter_ns()
a = np.arange(1024, dtype=np.float64).reshape(32,32)
value = int((a @ a.T).sum())`,8575959040},
 }
 for _,c:=range cases{for i:=0;i<4;i++{
 source:="import time\nbegin = time.perf_counter_ns()\n"+c.body+"\nfinish = time.perf_counter_ns()\nresult = {'value':value, 'import_ns':import_done-begin, 'compute_ns':finish-import_done}"
 start:=time.Now();out,err:=r.Run(ctx,source,nil);elapsed:=time.Since(start);if err!=nil{t.Fatal(c.name,err)}
 var v struct{Value int64 `json:"value"`;Import int64 `json:"import_ns"`;Compute int64 `json:"compute_ns"`};if err=json.Unmarshal(out.Value,&v);err!=nil{t.Fatal(err)}
 if v.Value!=c.want{t.Fatalf("%s got=%d want=%d",c.name,v.Value,c.want)}
 row:=map[string]any{"case":c.name,"sample":i,"warmup":i==0,"run_ns":elapsed.Nanoseconds(),"import_ns":v.Import,"compute_ns":v.Compute,"value":v.Value};b,_:=json.Marshal(row);fmt.Fprintln(os.Stdout,string(b))
 }}
}
